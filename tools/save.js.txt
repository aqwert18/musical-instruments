/* ---------- 正式版：紀錄與照片存到 Google Drive（覆寫樣板中同名的函式） ---------- */
async function persistRecord(rec,photos,progress){
 const saved=[];
 for(let k=0;k<photos.length;k++){
  const p=photos[k];
  if(!p.fileId){
   progress(`上傳照片 ${k+1}／${photos.length}…`);
   const blob=await compressPhoto(p.file);
   p.fileId=await upload({name:`${rec.date}_${rec.inst}_${rec.id}_${k+1}.jpg`,parents:[DB.photos],mimeType:'image/jpeg'},blob);
   if(p.url)photoCache[p.fileId]=Promise.resolve(p.url);
  }
  saved.push({caption:p.caption,fileId:p.fileId});
 }
 rec.photos=saved;
 progress('儲存紀錄…');
 const old=state.editing;
 await saveRecordToDrive(rec);
 // Photos removed while editing go to the Drive trash (recoverable for 30 days).
 if(old){const keep=new Set(saved.map(x=>x.fileId));(old.photos||[]).forEach(x=>{if(x.fileId&&!keep.has(x.fileId))trashFile(x.fileId).catch(()=>{})})}
}
async function removeRecord(rec){
 const latest=await readDB();
 DB.records=latest.filter(x=>x.id!==rec.id).sort((a,b)=>b.date.localeCompare(a.date));
 await upload({mimeType:'application/json'},dbBlob(),DB.file);
 (rec.photos||[]).forEach(p=>{if(p.fileId)trashFile(p.fileId).catch(()=>{})});
}
async function trashFile(id){
 const r=await gfetch(`${DRIVE}/${id}?fields=id`,{method:'PATCH',headers:{'Content-Type':'application/json'},body:JSON.stringify({trashed:true})});
 if(!r.ok)throw await driveErr(r,'刪除照片');
}
async function photoSrc(r,k){const p=(r.photos||[])[k];return p&&p.fileId?getPhotoUrl(p.fileId):''}
